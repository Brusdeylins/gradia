#type sequence

client:  "Web Client"     [ type: ACTOR ]
gateway: "API Gateway"    [ type: BOUNDARY ]
service: "Order Service"  [ type: CONTROL ]
order:   "Order"          [ type: ENTITY ]
db:      "Order Database" [ type: ENTITY ]
audit:   "Audit Log"

client  --("POST /orders")--> gateway { time: "t=now" }
gateway --("validate token")--> gateway
gateway --("token valid")--> gateway { kind: return }
gateway --(createOrder)--> service { duration: "{< 300ms}" }
service --("new Order")--> order { kind: create }
order   --(order)--> service { kind: return }
service --("INSERT order")--> db
db      --("order id")--> service { kind: return }
service --(dispose)--> order { kind: destroy }
service --("order created")--> audit { kind: async }
service --("201 Created")--> gateway { kind: return }
gateway --("201 Created")--> client { kind: return }
